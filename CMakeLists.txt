cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(HamtaroNative VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Hamtaro Native currently targets Windows x64.")
endif()
include(CTest)
include(cmake/Dependencies.cmake)
add_library(hn_core STATIC
    src/gba/memory.c src/gba/bios.c src/gba/bios_math.c src/gba/video.c src/gba/timing.c src/gba/timer.c src/gba/audio.c src/gba/psg.c
    src/game/input.c src/game/random.c src/game/heap.c src/game/script.c src/game/scene.c src/game/scene_motion.c src/game/text.c src/game/book.c src/game/actor.c src/game/motion.c src/game/collision.c src/game/runtime.c
    src/game/video.c src/game/transfer.c src/game/world.c src/game/expression.c src/game/save.c src/game/save_group3.c src/game/records.c src/game/record_menu.c src/game/counter_scene.c src/game/task.c src/game/task_text.c src/game/choice.c
    src/game/render.c src/game/presets.c src/game/actor_presets.c src/game/map_choice.c src/game/particles.c src/game/bank2.c src/game/bank3.c src/game/bank3_games.c src/game/bank3_chain.c src/game/bank4.c src/game/bank4_presets.c src/game/bank4_path.c src/game/wander.c src/game/scene_actor.c src/game/animation.c src/game/draw.c src/game/background.c src/game/graphics.c src/game/raster.c
    src/game/bank5.c src/game/bank5_presets.c
    src/game/bank6.c src/game/bank6_presets.c src/game/bank6_targets.c src/game/bank6_chain.c
    src/game/presentation.c src/game/presentation_background.c src/game/presentation_window.c
    src/game/audio.c src/game/audio_sequence.c src/game/audio_note.c src/game/audio_player.c
    src/game/audio_psg.c src/game/audio_mixer.c)
target_include_directories(hn_core PUBLIC src)
add_library(hn_storage STATIC
    src/app/storage.cpp src/app/settings.cpp src/app/rom.cpp)
target_include_directories(hn_storage PUBLIC src)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/rom.json" HN_ROM_SPEC)
string(JSON HN_ROM_SHA256 GET "${HN_ROM_SPEC}" sha256)
configure_file(cmake/rom_spec.h.in generated/rom_spec.h @ONLY)
target_include_directories(hn_storage PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(hn_storage PUBLIC bcrypt shell32 ole32 SDL3::Headers)
add_library(hn_platform STATIC src/platform/input.cpp src/platform/audio.cpp)
target_include_directories(hn_platform PUBLIC src)
target_link_libraries(hn_platform PUBLIC SDL3::SDL3-static hn_storage hn_core)
add_library(hn_session STATIC src/app/session.cpp)
target_include_directories(hn_session PUBLIC src)
target_link_libraries(hn_session PUBLIC hn_storage hn_core)
add_executable(hamtaro-native WIN32 src/app/main.cpp)
target_link_libraries(hamtaro-native PRIVATE hn_platform hn_imgui hn_session)
if(MSVC)
    foreach(target hn_core hn_storage hn_platform hn_session hamtaro-native)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    endforeach()
endif()
if(BUILD_TESTING)
    get_target_property(HN_CORE_SOURCES hn_core SOURCES)
    add_library(hn_reference_probe SHARED ${HN_CORE_SOURCES} tests/integration/reference_probe.c)
    target_include_directories(hn_reference_probe PRIVATE src)
    set_target_properties(hn_reference_probe PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    add_executable(hn_core_tests tests/unit/core.c)
    target_link_libraries(hn_core_tests PRIVATE hn_core)
    add_test(NAME core COMMAND hn_core_tests)
    add_executable(hn_game_tests tests/unit/game.c)
    target_link_libraries(hn_game_tests PRIVATE hn_core)
    add_test(NAME game COMMAND hn_game_tests)
    add_executable(hn_render_tests tests/unit/render.c)
    target_link_libraries(hn_render_tests PRIVATE hn_core)
    add_test(NAME render COMMAND hn_render_tests)
    add_executable(hn_graphics_tests tests/unit/graphics.c)
    target_link_libraries(hn_graphics_tests PRIVATE hn_core)
    add_test(NAME graphics COMMAND hn_graphics_tests)
    add_executable(hn_audio_tests tests/unit/audio.c)
    target_link_libraries(hn_audio_tests PRIVATE hn_core)
    add_test(NAME audio COMMAND hn_audio_tests)
    add_executable(hn_apu_tests tests/unit/apu.c)
    target_link_libraries(hn_apu_tests PRIVATE hn_core)
    add_test(NAME apu COMMAND hn_apu_tests)
    add_executable(hn_storage_tests tests/unit/storage.cpp)
    target_link_libraries(hn_storage_tests PRIVATE hn_storage)
    add_test(NAME storage COMMAND hn_storage_tests)
    add_executable(hn_input_tests tests/integration/input.cpp)
    target_link_libraries(hn_input_tests PRIVATE hn_platform)
    add_test(NAME input COMMAND hn_input_tests)
    add_executable(hn_session_tests tests/integration/session.cpp)
    target_link_libraries(hn_session_tests PRIVATE hn_session hn_platform)
    add_test(NAME session COMMAND hn_session_tests)
    add_test(NAME app-smoke COMMAND hamtaro-native --smoke-test)
    set_tests_properties(app-smoke PROPERTIES TIMEOUT 30)
    if(MSVC)
        foreach(target hn_reference_probe hn_core_tests hn_game_tests hn_render_tests hn_graphics_tests hn_audio_tests hn_apu_tests hn_storage_tests hn_input_tests hn_session_tests)
            target_compile_options(${target} PRIVATE /W4 /utf-8)
        endforeach()
    endif()
endif()
install(TARGETS hamtaro-native RUNTIME DESTINATION .)
install(FILES README.md NOTICE.md DESTINATION .)
install(FILES docs/architecture.md DESTINATION docs)
install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME SDL.txt)
install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME Dear-ImGui.txt)
install(FILES licenses/mGBA-MPL-2.0.txt DESTINATION licenses)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "hamtaro-native-${PROJECT_VERSION}-windows-x64")
include(CPack)
